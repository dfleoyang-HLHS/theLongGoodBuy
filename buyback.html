strategy_full_ui_v3.6.pine Pine Script v6 
//@version=6
strategy("回後買上漲 + 盤整突破 雙策略 (Pine v6) - 完整狀態儀表版 v3.6", overlay = true, initial_capital = 100000, default_qty_type = strategy.percent_of_equity, default_qty_value = 20, commission_type = strategy.commission.percent, commission_value = 0.2925, slippage = 1)

// ============================================================================
// 1. 輸入參數設定 (Inputs)
// ============================================================================
bool useDateFilter = input.bool(true,       title = "啟用回測時間區間篩選", group = "回測時間設定")
int startDate      = input.time(timestamp("2023-01-01 00:00 +0000"), title = "開始時間", group = "回測時間設定")
int endDate        = input.time(timestamp("2026-12-31 23:59 +0000"), title = "結束時間", group = "回測時間設定")

bool inDateRange = not useDateFilter or (time >= startDate and time <= endDate)

bool useStrategy1 = input.bool(true, title = "啟用策略1 (盤整突破)", group = "策略開關")
bool useStrategy2 = input.bool(true, title = "啟用策略2 (回後買上漲)", group = "策略開關")

int ma5Len          = input.int(5,     title = "短均線 (5 MA)", minval = 1, group = "均線設定")
int ma20Len         = input.int(20,    title = "月線 (20 MA)", minval = 1, group = "均線設定")
int trendLookback   = input.int(5,     title = "趨勢判斷回溯天數", minval = 2, group = "均線設定")

int lookbackLow     = input.int(3,     title = "停損防守 K 線數", minval = 1, group = "停損設定")
int lookbackHigh    = input.int(10,    title = "前高壓力區回溯 K 棒數", minval = 2, group = "停利設定")
float trailPerc     = input.float(3.5, title = "移動停利拉回百分比 (%)", step = 0.5, group = "停利設定")

int   consolidationLookback         = input.int(20,  title = "盤整區間回溯K棒數", minval = 5, group = "策略1-盤整突破設定")
float consolidationTightnessPercent = input.float(8, title = "盤整區間寬度上限(%)", step = 0.5, minval = 1, group = "策略1-盤整突破設定")

int   dipLookbackBars       = input.int(5,   title = "回檔觀察期間", minval = 1, group = "策略2-回後買上漲設定")
int   swingLeftBars         = input.int(5,   title = "前低偵測-左側K棒數", minval = 1, group = "策略2-回後買上漲設定")
int   swingRightBars        = input.int(5,   title = "前低偵測-右側確認K棒數", minval = 1, group = "策略2-回後買上漲設定")
float ma20UndershootPercent = input.float(1.0, title = "20MA支撐容忍度 (%)", step = 0.5, minval = 0, group = "策略2-回後買上漲設定")
float gainThresholdPercent  = input.float(2.0, title = "進場訊號紅K最小漲幅 (%)", step = 0.5, minval = 0, group = "策略2-回後買上漲設定")

bool  useVolumeFilter = input.bool(false, title = "啟用成交量加強濾網", group = "量能加強濾網")
int   volMaLen        = input.int(5,     title = "成交量均線週期", minval = 1, group = "量能加強濾網")
float volMult         = input.float(1.0, title = "成交量放大倍數", step = 0.1, group = "量能加強濾網")

bool   useVixFilter = input.bool(true,        title = "啟用 VIX 恐慌濾網", group = "雙VIX濾網設定")
string usVixSymbol  = input.symbol("CBOE:VIX", title = "美股 VIX 標的", group = "雙VIX濾網設定")
string twVixSymbol  = input.symbol("TAIFEX:VI",title = "台股 VIX 標的", group = "雙VIX濾網設定")
float  vixThreshold = input.float(20,         title = "VIX 恐慌門檻", step = 1, minval = 1, group = "雙VIX濾網設定")
string vixLogic     = input.string("AND",     title = "雙VIX判定條件", options = ["AND", "OR"], group = "雙VIX濾網設定")

int barsPerYearApprox = input.int(252, title = "回溯K棒數 (近一年)", minval = 1, group = "狀態顯示設定")

// ============================================================================
// 2. 指標計算
// ============================================================================
series float ma5   = ta.sma(close, ma5Len)
series float ma20  = ta.sma(close, ma20Len)
series float volMa = ta.sma(volume, volMaLen)

float usVixValue = request.security(usVixSymbol, timeframe.period, close[1], lookahead = barmerge.lookahead_on)
float twVixValue = request.security(twVixSymbol, timeframe.period, close[1], lookahead = barmerge.lookahead_on)

bool isUsVixCalm = usVixValue < vixThreshold
bool isTwVixCalm = twVixValue < vixThreshold
bool isVixCalm   = not useVixFilter or (vixLogic == "AND" ? (isUsVixCalm and isTwVixCalm) : (isUsVixCalm or isTwVixCalm))

bool isVolumeConfirmed = not useVolumeFilter or volume > (volMa * volMult)
bool isBullTrend       = close > ma20 and ma20 > ma20[trendLookback]

// ============================================================================
// 3. 策略條件
// ============================================================================
// 策略1：盤整突破
float rangeHigh = ta.highest(high, consolidationLookback)[1]
float rangeLow  = ta.lowest(low, consolidationLookback)[1]
bool  isConsolidating    = (rangeHigh - rangeLow) / rangeLow * 100 <= consolidationTightnessPercent
bool  breakoutAboveRange = close > rangeHigh
bool  strategy1Condition = useStrategy1 and isBullTrend and isConsolidating and breakoutAboveRange

// 策略2：回後買上漲
float pivotLowRaw = ta.pivotlow(swingLeftBars, swingRightBars)
var float lastConfirmedSwingLow = na
if not na(pivotLowRaw)
    lastConfirmedSwingLow := pivotLowRaw

bool notBreakPriorLow    = na(lastConfirmedSwingLow) or ta.lowest(low, dipLookbackBars) >= lastConfirmedSwingLow
bool notBreakMa20        = ta.lowest(low, dipLookbackBars) >= ma20 * (1 - ma20UndershootPercent / 100)
bool wasBelow5MARecently = ta.barssince(close < ma5) <= dipLookbackBars
bool reclaim5MA          = close > ma5 and wasBelow5MARecently
float pctChangeToday     = (close - close[1]) / close[1] * 100
bool strongRedCandle     = close > open and pctChangeToday >= gainThresholdPercent
bool strategy2Condition  = useStrategy2 and isBullTrend and notBreakPriorLow and notBreakMa20 and reclaim5MA and strongRedCandle

// 綜合進場條件
bool buyConditionStrategy1 = strategy1Condition and isVolumeConfirmed and isVixCalm and inDateRange
bool buyConditionStrategy2 = strategy2Condition and isVolumeConfirmed and isVixCalm and inDateRange
bool buyCondition          = buyConditionStrategy1 or buyConditionStrategy2

string entryStrategyLabel  = buyConditionStrategy1 ? "策略1(盤整突破)" : buyConditionStrategy2 ? "策略2(回後買上漲)" : ""
float recentSignalCount    = math.sum(buyCondition ? 1 : 0, barsPerYearApprox)

// ============================================================================
// 4. 策略執行邏輯
// ============================================================================
series float lowestLow   = ta.lowest(low, lookbackLow)
series float highestHigh = ta.highest(high, lookbackHigh)

var float stopLossPrice  = na
var float takeProfitHigh = na
var float maxPostHigh    = na
var float trailingStop   = na
var bool  tpTaken        = false

if (buyCondition and strategy.position_size == 0)
    stopLossPrice  := lowestLow
    takeProfitHigh := highestHigh[1]
    maxPostHigh    := high
    trailingStop   := high * (1 - trailPerc / 100)
    tpTaken        := false
    strategy.entry("Long", strategy.long)

if (strategy.position_size > 0)
    if (high > maxPostHigh)
        maxPostHigh  := high
        trailingStop := maxPostHigh * (1 - trailPerc / 100)

    if (not tpTaken and strategy.position_size[1] > strategy.position_size)
        tpTaken := true

    if (close < ma20)
        strategy.close("Long", comment = "跌破 20 MA 離場")

if (strategy.position_size > 0)
    float combinedStop = math.max(stopLossPrice, trailingStop)
    strategy.exit("保護停損", from_entry = "Long", stop = combinedStop)
    if (not tpTaken)
        strategy.exit("前高停利", from_entry = "Long", qty_percent = 50, limit = takeProfitHigh)

// ============================================================================
// 5. 補回：圖表繪製、進場標籤與右上狀態儀表板 (UI Components)
// ============================================================================
plot(ma5,  title = "5 MA",  color = color.yellow, linewidth = 1)
plot(ma20, title = "20 MA", color = color.blue,   linewidth = 2)

// K 線圖下方的買进訊號標籤 (Strategy 1 & 2)
plotshape(buyConditionStrategy1, title = "策略1買進標籤", style = shape.labelup, location = location.belowbar,
          color = color.teal, text = "策略1(突破)", textcolor = color.white, size = size.normal)
plotshape(buyConditionStrategy2, title = "策略2買進標籤", style = shape.labelup, location = location.belowbar,
          color = color.green, text = "策略2(回後買)", textcolor = color.white, size = size.normal)

// 觸發買進當日 K 線背景反白
bgcolor(buyConditionStrategy1 ? color.new(color.teal, 85) : buyConditionStrategy2 ? color.new(color.green, 85) : na, title = "買進訊號日背景反白")

// 停損停利線
plot(strategy.position_size > 0 ? stopLossPrice : na,  title = "初始停損價線", color = color.red,    style = plot.style_linebr, linewidth = 2)
plot(strategy.position_size > 0 ? takeProfitHigh : na, title = "前高停利價線", color = color.purple, style = plot.style_circles, linewidth = 2)
plot(strategy.position_size > 0 ? trailingStop : na,   title = "移動停利價線", color = color.green,  style = plot.style_linebr, linewidth = 2)

// 右上方每日判斷表格 (Status Table)
var table statusTable = table.new(position.top_right, 2, 10, border_width = 1)
if barstate.islast
    color okColor = color.new(color.green, 0)
    color noColor = color.new(color.red, 0)

    table.cell(statusTable, 0, 0, "雙策略每日判斷狀態", text_color = color.white, bgcolor = color.navy, text_size = size.tiny)
    table.cell(statusTable, 1, 0, "近" + str.tostring(barsPerYearApprox) + "棒觸發次數: " + str.tostring(recentSignalCount, "#"), text_color = color.white, bgcolor = color.navy, text_size = size.tiny)

    table.cell(statusTable, 0, 1, "多頭趨勢 (共用前提)", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 1, isBullTrend ? "符合" : "不符合", text_color = color.white, bgcolor = isBullTrend ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 2, "美VIX:" + str.tostring(usVixValue, "#.#") + " / 台VIX:" + str.tostring(twVixValue, "#.#"), text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 2, isVixCalm ? "市場平穩" : "恐慌偏高", text_color = color.white, bgcolor = isVixCalm ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 3, "【策略1】是否處於盤整", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 3, isConsolidating ? "符合" : "不符合", text_color = color.white, bgcolor = isConsolidating ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 4, "【策略1】是否突破區間高點", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 4, breakoutAboveRange ? "符合" : "不符合", text_color = color.white, bgcolor = breakoutAboveRange ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 5, "【策略2】條件1-不破前低", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 5, notBreakPriorLow ? "符合" : "不符合", text_color = color.white, bgcolor = notBreakPriorLow ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 6, "【策略2】條件2-不破月線", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 6, notBreakMa20 ? "符合" : "不符合", text_color = color.white, bgcolor = notBreakMa20 ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 7, "【策略2】站回5日線", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 7, reclaim5MA ? "符合" : "不符合", text_color = color.white, bgcolor = reclaim5MA ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 8, "【策略2】紅K漲幅≥" + str.tostring(gainThresholdPercent, "#.#") + "%", text_color = color.white, bgcolor = color.gray, text_size = size.tiny)
    table.cell(statusTable, 1, 8, strongRedCandle ? "符合" : "不符合", text_color = color.white, bgcolor = strongRedCandle ? okColor : noColor, text_size = size.tiny)

    table.cell(statusTable, 0, 9, "① 是否該進場買進", text_color = color.white, bgcolor = color.black, text_size = size.tiny)
    string buyStatusText = buyConditionStrategy1 ? "✅ 策略1觸發" : buyConditionStrategy2 ? "✅ 策略2觸發" : "❌ 尚未符合"
    table.cell(statusTable, 1, 9, buyStatusText, text_color = color.white, bgcolor = buyCondition ? okColor : noColor, text_size = size.tiny)
